     1	// textbuf.h —— 文本缓冲区:行数组 + 逆操作日志(撤销/重做)
     2	//
     3	// 为什么是行数组:单文件竞赛源码不过几千行,插入/删除行的 O(n) memmove
     4	// 完全不可感知;gap buffer / piece table 换不来任何用户可感收益,而行数组
     5	// 让语法高亮、渲染、行号跳转都变直接。
     6	//
     7	// 核心契约:
     8	//   * lines_ 永远至少 1 行(空文件 = 一个空行)。
     9	//   * **唯一两个改动存储的原语是 insert / erase**,二者都记录撤销。
    10	//     §6 的“练习模式绝不插入缓冲区”正是建立在这个咽喉点唯一性之上:
    11	//     AI 文本到达它们的路径只有 Editor::acceptGhost() 一条。
    12	//   * Pos::col 是**行内字节偏移**,不是显示列。显示列换算见 util.h。
    13	//   * 所有只读访问器越界安全(line(i) 越界返回静态空串)。
    14	//   * TextBuffer 不是线程安全的,只在 UI 线程使用;工作线程只拿到它的快照文本。
    15	#pragma once
    16	
    17	#include <cstdint>
    18	#include <string>
    19	#include <vector>
    20	
    21	// 缓冲区内的位置:行号(0 起)+ 行内字节偏移。
    22	struct Pos {
    23	  int line = 0;
    24	  int col = 0;
    25	};
    26	bool operator==(const Pos& a, const Pos& b);
    27	bool operator!=(const Pos& a, const Pos& b);
    28	bool operator<(const Pos& a, const Pos& b);
    29	bool operator<=(const Pos& a, const Pos& b);
    30	
    31	// 半开区间 [a, b),约定 a <= b(用 normalized() 规范化)。
    32	struct Range {
    33	  Pos a, b;
    34	  bool empty() const { return a == b; }
    35	  Range normalized() const { return (a <= b) ? Range{a, b} : Range{b, a}; }
    36	};
    37	
    38	enum class Lang : uint8_t { C, Cpp, Unknown };
    39	
    40	// 撤销日志的一条逆操作。isInsert==true 表示“当初插入了 text”,
    41	// 撤销它就是删掉;false 表示“当初删除了 text”,撤销它就是插回去。
    42	struct UndoOp {
    43	  bool isInsert = true;
    44	  Pos at;
    45	  std::string text;
    46	};
    47	
    48	// 一次用户动作 = 一个组。所以“接受 ghost 插入 8 行”一次撤销就回退干净。
    49	struct UndoGroup {
    50	  std::vector<UndoOp> ops;
    51	  Pos cursorBefore, cursorAfter;
    52	  int64_t stamp = 0;              // util::nowMs(),用于 800ms 合并窗口
    53	  const char* label = nullptr;    // 静态字符串字面量,不拥有
    54	};
    55	
    56	class TextBuffer {
    57	 public:
    58	  TextBuffer();
    59	
    60	  // 整体替换内容(清空撤销栈、revision++、dirty 置 false)。lines 为空时补一个空行。
    61	  void reset(std::vector<std::string> lines);
    62	  // 读文件。失败时 err 为中文说明。成功后 path()/lang() 更新,dirty 置 false。
    63	  bool loadFile(const std::string& path, std::string& err);
    64	  // 原子写:同目录 tmp + rename。成功后 clearDirty()。
    65	  bool saveFile(const std::string& path, std::string& err);
    66	
    67	  int lineCount() const;
    68	  const std::string& line(int i) const;     // 越界安全,返回静态空行
    69	  int lineLen(int i) const;                 // 字节数;越界返回 0
    70	  const std::vector<std::string>& lines() const;
    71	  std::string text() const;                 // 各行用 '\n' 连接,末尾补一个 '\n'
    72	  std::string textRange(const Range& r) const;
    73	  // 位置工具
    74	  Pos clampPos(Pos p) const;                // 夹到合法范围(并对齐到 UTF-8 字符边界)
    75	  Pos endPos() const;                       // 最后一行行尾
    76	  bool hasNonBlank() const;                 // 是否存在至少一个非空白字符(AI 自动触发条件)
    77	
    78	  // ---- 唯一两个改动存储的原语,二者都记录撤销 ----
    79	  Pos insert(Pos at, const std::string& text);  // 处理 '\n' 拆行,返回结束位置
    80	  Pos erase(const Range& r);                    // 返回规范化后的 r.a
    81	
    82	  // ---- 撤销分组 ----
    83	  // beginGroup/endGroup 可嵌套(只有最外层真正开/关组)。label 必须是静态字符串。
    84	  void beginGroup(const char* label);
    85	  void endGroup();
    86	  // RAII 包装:一次用户动作写 `TextBuffer::Edit guard(buf, "insert-char");`
    87	  struct Edit {
    88	    Edit(TextBuffer& b, const char* label);
    89	    ~Edit();
    90	    Edit(const Edit&) = delete;
    91	    Edit& operator=(const Edit&) = delete;
    92	   private:
    93	    TextBuffer& b_;
    94	  };
    95	  // 可选:告诉缓冲区“动作前/后的光标在哪”,用于撤销后精确恢复光标。
    96	  // 不调用也能工作(退化为由 ops 位置推断)。Editor 在每次编辑前后各调一次。
    97	  void noteCursorBefore(Pos p);
    98	  void noteCursorAfter(Pos p);
    99	  // 强制关闭当前合并窗口:回车、光标移动、保存、剪切行等结构性操作要调它,
   100	  // 否则连续单字符插入的合并规则会把不相关的动作粘成一组。
   101	  void breakUndoMerge();
   102	
   103	  bool undo(Pos& cursorOut);
   104	  bool redo(Pos& cursorOut);
   105	  bool canUndo() const;
   106	  bool canRedo() const;
   107	  void clearHistory();
   108	
   109	  bool dirty() const;
   110	  void clearDirty();
   111	  int revision() const;                     // 每次改动 +1 -> 高亮/AI 失效判据
   112	  const std::string& path() const;
   113	  void setPath(std::string p);              // 同时按扩展名重推 lang
   114	  Lang lang() const;
   115	  static Lang langFromPath(const std::string& path);
   116	
   117	 private:
   118	  void recordOp(UndoOp op);
   119	  void applyOpNoJournal(const UndoOp& op, Pos& cursorOut);
   120	  Pos rawInsert(Pos at, const std::string& text);
   121	  std::string rawErase(const Range& r);
   122	
   123	  std::vector<std::string> lines_;
   124	  std::string path_;
   125	  Lang lang_ = Lang::Unknown;
   126	  bool dirty_ = false;
   127	  int revision_ = 0;
   128	
   129	  // 撤销状态
   130	  bool journaling_ = true;                  // 撤销/重做期间挂起记录
   131	  int group_depth_ = 0;
   132	  bool group_open_ = false;                 // 当前是否有一个“可合并的开着的组”
   133	  const char* cur_label_ = nullptr;
   134	  int64_t last_edit_ms_ = 0;
   135	  Pos pending_before_{};
   136	  bool have_pending_before_ = false;
   137	  std::vector<UndoGroup> undo_;
   138	  std::vector<UndoGroup> redo_;
   139	};
