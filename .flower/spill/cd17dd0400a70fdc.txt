# 有没有命令行里含某个模式的活进程?直接读 /proc(本容器里既没有 pgrep 也没有 ps)。
# macOS 上没有 /proc,那里会退回用 pgrep;两者都没有就打印"跳过"。
orphans() { # orphans <模式>
  if [ -d /proc ]; then
    python3 - "$1" <<'PY'
import os, sys
pat = sys.argv[1].encode()
me = os.getpid()
hits = []
for d in os.listdir('/proc'):
    if not d.isdigit() or int(d) == me:
        continue
    try:
        with open('/proc/%s/cmdline' % d, 'rb') as f:
            cl = f.read()
    except OSError:
        continue
    if pat in cl:
        hits.append(d + ' ' + cl.replace(b'\0', b' ').decode('utf-8', 'replace')[:120])
for h in hits:
    print(h)
sys.exit(1 if hits else 0)
PY
    return $?
  fi
  if command -v pgrep > /dev/null 2>&1; then pgrep -f "$1" > /dev/null 2>&1; return $?; fi
  return 2   # 无法判断
}

check_no_orphans() { # check_no_orphans <模式> <说明>
  out=$(orphans "$1"); rc=$?
  case $rc in
    0) ok "$2:没有残留进程" ;;
    1) bad "$2:仍有残留进程 -> $out" ;;
    *) printf '  跳过 %s 的孤儿进程检查(既无 /proc 也无 pgrep)\n' "$2" ;;
  esac
}

drive() { # drive <rows> <cols> <script> <out> <bin> <cfg> <file> [超时秒]
  r=$1; c=$2; sc=$3; out=$4; b=$5; cfg=$6; f=$7; to=${8:-300}
  CPPIDE_API_KEY= DEEPSEEK_API_KEY= \
    timeout "$to" "$PTY" "$r" "$c" "$sc" "$out" -- "$b" --config "$cfg" "$f"
}
