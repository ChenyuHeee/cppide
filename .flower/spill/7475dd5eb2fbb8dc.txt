120	    return false;
121	  }
122	  if (!positional.empty()) {
123	    if (positional[0].empty()) {
124	      err = "文件名为空。";
125	      return false;
126	    }
127	    a.file = util::expandUser(positional[0]);
128	  }
129	  return true;
130	}
