255	  full.clear();
256	  bool cxx_ok = findInPath(cfg.cxx, full);
257	  if (!cc_ok || !cxx_ok) {
258	    r += "- 编译器未找到:请安装 Xcode Command Line Tools(xcode-select --install),"
259	         "或在配置里把 cc/cxx 改成实际路径。\n";
260	  } else {
261	    r += "- 编译与运行功能就绪。\n";
262	  }
263	  r += cfg.aiEnabled() ? "- AI 功能已启用。\n"
264	                       : "- AI 功能未启用(api_key 为空);编辑、编译、运行均不受影响。\n";
265	  return r;
266	}
267	
268	// -------------------------------------------------------- 打开文件的前置检查
269	// 不存在 / 是目录 / 读不了 都要给中文报错并以非 0 退出,绝不带着坏状态进 TUI。
270	bool checkOpenPath(const std::string& p, std::string& err) {
271	  if (util::isDirectory(p)) {
272	    err = p + " 是一个目录,不是文件。";
273	    return false;
274	  }
275	  if (!util::fileExists(p)) {
276	    err = "找不到文件 " + p + "。\n"
277	          "       提示:不带参数启动 cppide 会开一个空缓冲区,编辑后用 Ctrl-O 另存为。";
278	    return false;
279	  }
280	  std::string text, rerr;
281	  if (!util::readFile(p, text, rerr)) {
282	    err = "无法读取 " + p + ":" + rerr;
283	    return false;
284	  }
285	  return true;
286	}
